<script>
    const BASE_SPEED = 0.4;
    const CANVAS_WIDTH = 1200;
    const CANVAS_HEIGHT = 600;

    function getCSSVar(name) {
        return getComputedStyle(document.documentElement).getPropertyValue(name).trim();
    }

    function generateCloudSVG(color) {
        const svgString = `
            <svg xmlns="http://www.w3.org/2000/svg" width="200" height="100" viewBox="0 0 200 100">
                <path fill="${color}" d="M40 50h120v15H40z M60 35h80v15H60z M80 20h40v15H80z M30 60h140v10H30z" />
            </svg>`.trim();
        return `url('data:image/svg+xml;base64,${window.btoa(svgString)}')`;
    }

    function generateMountainSVG(colorLight, colorDark, peaks, roughness, index) {
        const gradId = `grad-${index}`;
        let path = `M 0 ${CANVAS_HEIGHT}`;
        const step = CANVAS_WIDTH / peaks;
        let startY = 250 + Math.random() * 100;
        if(index === 3) startY = 180; 
        
        path += ` L 0 ${startY}`;
        let currentY = startY;
        let highestPoint = startY;

        for (let i = 1; i <= peaks; i++) {
            let x = i * step;
            if (i === peaks) {
                currentY = startY;
            } else {
                let change = (Math.random() - 0.5) * roughness * 6;
                currentY += change;
                currentY = Math.max(50, Math.min(CANVAS_HEIGHT - 20, currentY));
                if (currentY < highestPoint) highestPoint = currentY;
            }
            path += ` L ${x} ${currentY}`;
        }
        path += ` L ${CANVAS_WIDTH} ${CANVAS_HEIGHT} Z`;

        const svgString = `
            <svg xmlns="http://www.w3.org/2000/svg" width="${CANVAS_WIDTH}" height="${CANVAS_HEIGHT}">
                <defs>
                    <linearGradient id="${gradId}" x1="0" y1="${CANVAS_HEIGHT}" x2="0" y2="${highestPoint}" gradientUnits="userSpaceOnUse">
                        <stop offset="0%" stop-color="${colorDark}" />
                        <stop offset="100%" stop-color="${colorLight}" />
                    </linearGradient>
                </defs>
                <path fill="url(#${gradId})" d="${path}"/>
            </svg>`;
        return `url('data:image/svg+xml;base64,${window.btoa(svgString)}')`;
    }

    // Configuração das Nuvens
    const cloudColor = getCSSVar('--cloud-color');
    const cloudSVG = generateCloudSVG(cloudColor);
    
    class Cloud {
        constructor(container, speedMult, className) {
            this.el = document.createElement('div');
            this.el.className = `cloud-item ${className}`;
            this.el.style.backgroundImage = cloudSVG;
            this.speedMult = speedMult;
            container.appendChild(this.el);
            this.reset(true);
        }
        reset(isFirstTime = false) {
            this.width = 150 + Math.random() * 150;
            this.height = this.width / 2;
            this.el.style.width = `${this.width}px`;
            this.el.style.height = `${this.height}px`;
            this.x = isFirstTime ? Math.random() * window.innerWidth : window.innerWidth + 200;
            this.y = (window.innerHeight * 0.05) + Math.random() * (window.innerHeight * 0.35);
            this.update();
        }
        update() {
            this.x -= BASE_SPEED * this.speedMult;
            if (this.x < -this.width) this.reset();
            this.el.style.transform = `translate(${this.x}px, ${this.y}px)`;
        }
    }

    const cloudsBack = [];
    const cloudsFront = [];
    for(let i=0; i<3; i++) {
        cloudsBack.push(new Cloud(document.getElementById('cloudsBackContainer'), 0.4, 'cloud-back-layer'));
        cloudsFront.push(new Cloud(document.getElementById('cloudsFrontContainer'), 1.2, 'cloud-front-layer'));
    }

    const layers = document.querySelectorAll('.layer');
    layers.forEach((layer, index) => {
        const cLight = getCSSVar(layer.getAttribute('data-light').replace(/var\(|\)/g, ''));
        const cDark = getCSSVar(layer.getAttribute('data-dark').replace(/var\(|\)/g, ''));
        layer.style.backgroundImage = generateMountainSVG(cLight, cDark, parseInt(layer.getAttribute('data-peaks')), parseInt(layer.getAttribute('data-roughness')), index);
        layer.style.backgroundSize = `${CANVAS_WIDTH}px 100%`;
    });

    // === ESTRELAS QUADRADAS (Ajustadas para 2x2 pixels) ===
    const starsContainer = document.getElementById('stars');
    for(let i = 0; i < 70; i++) {
        const star = document.createElement('div');
        star.className = 'star';
        // Tamanho fixo de 2px ou 4px para parecer pixel art
        const size = Math.random() < 0.8 ? 2 : 4; 
        star.style.width = size + 'px'; 
        star.style.height = size + 'px';
        star.style.left = Math.random() * 100 + '%'; 
        star.style.top = Math.random() * 100 + '%';
        star.style.animationDelay = Math.random() * 5 + 's';
        starsContainer.appendChild(star);
    }

    let scrollPosition = 0;
    function animate() {
        scrollPosition += BASE_SPEED;
        layers.forEach(layer => {
            const speed = parseFloat(layer.getAttribute('data-speed'));
            layer.style.backgroundPositionX = `-${scrollPosition * speed}px`;
        });
        cloudsBack.forEach(c => c.update());
        cloudsFront.forEach(c => c.update());
        requestAnimationFrame(animate);
    }
    animate();

    window.addEventListener('resize', () => {
        cloudsBack.forEach(c => { if(c.x > window.innerWidth) c.reset() });
        cloudsFront.forEach(c => { if(c.x > window.innerWidth) c.reset() });
    });
</script>
