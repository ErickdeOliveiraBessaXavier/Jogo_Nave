## 1. 🎚️ Sistema de Dificuldade Selecionável

### Conceito
Permite jogadores escolherem o nível de desafio antes de começar, como em jogos modernos (Easy/Normal/Hard/Nightmare).

### Implementação

```python
# Em levels.py

from enum import Enum

class DifficultyPreset(Enum):
    """Presets de dificuldade para o jogador escolher."""
    
    CASUAL = "casual"
    NORMAL = "normal"
    HARDCORE = "hardcore"
    NIGHTMARE = "nightmare"


class DifficultySettings:
    """Configurações específicas por preset de dificuldade."""
    
    PRESETS = {
        DifficultyPreset.CASUAL: {
            "name": "Casual",
            "description": "Relaxe e aproveite a jornada",
            "difficulty_scaling": 0.10,      # Dificuldade sobe mais devagar
            "spawn_rate_multiplier": 0.75,   # Menos inimigos por segundo
            "enemy_health_multiplier": 0.8,  # Inimigos mais fracos
            "player_damage_multiplier": 1.3, # Jogador mais forte
            "lives": 5,                      # Mais vidas
            "rewards_multiplier": 0.8,       # Menos recompensas
        },
        DifficultyPreset.NORMAL: {
            "name": "Normal",
            "description": "Experiência balanceada",
            "difficulty_scaling": 0.15,
            "spawn_rate_multiplier": 1.0,
            "enemy_health_multiplier": 1.0,
            "player_damage_multiplier": 1.0,
            "lives": 3,
            "rewards_multiplier": 1.0,
        },
        DifficultyPreset.HARDCORE: {
            "name": "Hardcore",
            "description": "Para veteranos",
            "difficulty_scaling": 0.22,
            "spawn_rate_multiplier": 1.4,
            "enemy_health_multiplier": 1.3,
            "player_damage_multiplier": 0.9,
            "lives": 2,
            "rewards_multiplier": 1.5,
        },
        DifficultyPreset.NIGHTMARE: {
            "name": "Pesadelo",
            "description": "Você foi avisado",
            "difficulty_scaling": 0.30,
            "spawn_rate_multiplier": 2.0,
            "enemy_health_multiplier": 1.5,
            "player_damage_multiplier": 0.8,
            "lives": 1,  # Permadeath!
            "rewards_multiplier": 3.0,  # Alta recompensa, alto risco

    @classmethod

        """Retorna configurações para um preset."""
        return cls.PRESETS[preset]

        curve = DifficultyConfig.SPAWN_RATE_CURVE
        scaling = self.settings["difficulty_scaling"]  # ← Agora vem do preset
        base = 1.0
        
        if curve == "logarithmic":
            difficulty = base + math.log1p(level_number) * scaling
```

## 2. 🎲 Modificadores de Run (Roguelike-Style)

### Conceito
Inspirado em Hades, Slay the Spire, Dead Cells - o jogador escolhe "maldições" que tornam o jogo mais difícil mas dão recompensas maiores.

### Implementação

```python
from dataclasses import dataclass
from typing import Callable

@dataclass
class RunModifier:
    """Modificador que afeta toda a run."""
    
    id: str
    name: str
    description: str
    icon: str
    
    # Multiplicadores
    spawn_rate_multiplier: float = 1.0
    enemy_speed_multiplier: float = 1.0
    enemy_health_multiplier: float = 1.0
    player_damage_multiplier: float = 1.0
    player_speed_multiplier: float = 1.0
    
    # Recompensas
    score_multiplier: float = 1.0
    xp_multiplier: float = 1.0
    currency_multiplier: float = 1.0
    
    # Mecânicas especiais
    special_rules: list[str] = None
    on_level_start: Callable | None = None
    on_enemy_spawn: Callable | None = None
    
    def __post_init__(self):
        if self.special_rules is None:
            self.special_rules = []


# Biblioteca de modificadores

class ModifierManager:
    """Gerencia modificadores ativos durante uma run."""
    
    def __init__(self):
        self.active_modifiers: list[RunModifier] = []
    
    def add_modifier(self, modifier_id: str):
        """Adiciona um modificador à run."""
        if modifier_id in MODIFIERS_LIBRARY:
            modifier = MODIFIERS_LIBRARY[modifier_id]
            self.active_modifiers.append(modifier)
    
    def get_total_multiplier(self, stat: str) -> float:
        """Calcula multiplicador total para uma stat."""
        total = 1.0
        for mod in self.active_modifiers:
            total *= getattr(mod, stat, 1.0)
        return total
    
    def has_special_rule(self, rule: str) -> bool:
        """Verifica se algum modificador tem uma regra especial."""
        return any(rule in mod.special_rules for mod in self.active_modifiers)
    
    def get_total_score_multiplier(self) -> float:
        """Multiplica todos os score multipliers."""
        return self.get_total_multiplier("score_multiplier")
    
    def apply_to_level_config(self, config: LevelConfig) -> LevelConfig:
        """Aplica modificadores a uma configuração de nível."""
        spawn_mult = self.get_total_multiplier("spawn_rate_multiplier")
        
        # Modificar spawn times
        new_spawn_config = {}
        for enemy_type, spawn_time in config.enemy_spawn_config.items():
            new_spawn_config[enemy_type] = spawn_time / spawn_mult
        
        # Boss Rush: forçar boss a cada 2 níveis
        boss_type = config.boss_type
        if self.has_special_rule("boss_every_2_levels"):
            if config.level_number % 2 == 0:
                boss_type = Boss  # Forçar boss
        
        return LevelConfig(
            level_number=config.level_number,
            enemy_spawn_config=new_spawn_config,
            enemies_to_clear=config.enemies_to_clear,
            boss_type=boss_type,
            mines_enabled=config.mines_enabled,
            formations_enabled=config.formations_enabled,
            formation_types=config.formation_types,
            theme_name=config.theme_name,
        )
```

## 3. 📊 Sistema de Meta-Progression

### Conceito
O jogo aprende com você e ajusta sutilmente a dificuldade. Inspirado em Left 4 Dead's "AI Director" e Resident Evil 4's adaptive difficulty.

### Implementação

```python
import json
from datetime import datetime, timedelta
from pathlib import Path

@dataclass
class LevelPerformance:
    """Estatísticas de performance em um nível."""
    level_number: int
    attempts: int = 0
    clears: int = 0
    deaths: int = 0
    total_time: float = 0.0  # segundos
    best_time: float = float('inf')
    avg_score: float = 0.0
    last_played: datetime = None
    
    @property
    def clear_rate(self) -> float:
        """Taxa de sucesso (0.0 a 1.0)."""
        return self.clears / self.attempts if self.attempts > 0 else 0.0
    
    @property
    def avg_time(self) -> float:
        """Tempo médio de clear."""
        return self.total_time / self.clears if self.clears > 0 else 0.0
    
    def is_struggling(self) -> bool:
        """Jogador está tendo dificuldade?"""
        return (
            self.attempts >= 3 and
            self.clear_rate < 0.4  # Menos de 40% de sucesso
        )
    
    def is_dominating(self) -> bool:
        """Jogador está achando fácil demais?"""
        return (
            self.attempts >= 2 and
            self.clear_rate > 0.9 and  # Mais de 90% de sucesso
            self.avg_time < self.best_time * 1.2  # Tempos consistentes
        )


class PlayerProfile:
    """Perfil do jogador com histórico e meta-progression."""
    
    def __init__(self, profile_path: Path):
        self.profile_path = profile_path
        self.level_stats: dict[int, LevelPerformance] = {}
        self.total_playtime: float = 0.0
        self.highest_level_reached: int = 1
        self.total_deaths: int = 0
        self.load()
    
    def load(self):
        """Carrega perfil do disco."""
        if not self.profile_path.exists():
            return
        
        try:
            with open(self.profile_path, 'r') as f:
                data = json.load(f)
                self.total_playtime = data.get('total_playtime', 0.0)
                self.highest_level_reached = data.get('highest_level_reached', 1)
                self.total_deaths = data.get('total_deaths', 0)
                
                for level_num, stats in data.get('level_stats', {}).items():
                    self.level_stats[int(level_num)] = LevelPerformance(**stats)
        except Exception as e:
            print(f"Erro ao carregar perfil: {e}")
    
    def save(self):
        """Salva perfil no disco."""
        data = {
            'total_playtime': self.total_playtime,
            'highest_level_reached': self.highest_level_reached,
            'total_deaths': self.total_deaths,
            'level_stats': {
                level_num: {
                    'level_number': stats.level_number,
                    'attempts': stats.attempts,
                    'clears': stats.clears,
                    'deaths': stats.deaths,
                    'total_time': stats.total_time,
                    'best_time': stats.best_time,
                    'avg_score': stats.avg_score,
                    'last_played': stats.last_played.isoformat() if stats.last_played else None,
                }
                for level_num, stats in self.level_stats.items()
            }
        }
        
        with open(self.profile_path, 'w') as f:
            json.dump(data, f, indent=2)
    
    def record_attempt(self, level_number: int):
        """Registra tentativa de um nível."""
        if level_number not in self.level_stats:
            self.level_stats[level_number] = LevelPerformance(level_number)
        
        self.level_stats[level_number].attempts += 1
        self.level_stats[level_number].last_played = datetime.now()
    
    def record_clear(self, level_number: int, time_taken: float, score: float):
        """Registra clear de um nível."""
        stats = self.level_stats[level_number]
        stats.clears += 1
        stats.total_time += time_taken
        stats.best_time = min(stats.best_time, time_taken)
        stats.avg_score = (stats.avg_score * (stats.clears - 1) + score) / stats.clears
        
        self.highest_level_reached = max(self.highest_level_reached, level_number + 1)
        self.save()
    
    def record_death(self, level_number: int):
        """Registra morte em um nível."""
        self.level_stats[level_number].deaths += 1
        self.total_deaths += 1
        self.save()
    
    def get_adjusted_config(self, base_config: LevelConfig) -> LevelConfig:
        """
        Ajusta configuração baseado em performance do jogador.
        
        Sistema sutil: ajustes de 5-15%, não óbvios para o jogador.
        """
        level_num = base_config.level_number
        
        if level_num not in self.level_stats:
            return base_config  # Primeira tentativa, sem ajuste
        
        stats = self.level_stats[level_num]
        adjustment = 1.0  # Multiplicador neutro
        
        # Jogador lutando? Facilitar sutilmente
        if stats.is_struggling():
            adjustment = 0.85  # -15% de dificuldade
            print(f"[Meta-Progression] Facilitando Level {level_num} (clear rate: {stats.clear_rate:.0%})")
        
        # Jogador dominando? Dificultar sutilmente
        elif stats.is_dominating():
            adjustment = 1.15  # +15% de dificuldade
            print(f"[Meta-Progression] Dificultando Level {level_num} (clear rate: {stats.clear_rate:.0%})")
        
        # Aplicar ajuste aos spawn times
        adjusted_spawn_config = {}
        for enemy_type, spawn_time in base_config.enemy_spawn_config.items():
            adjusted_spawn_config[enemy_type] = spawn_time / adjustment
        
        return LevelConfig(
            level_number=base_config.level_number,
            enemy_spawn_config=adjusted_spawn_config,
            enemies_to_clear=int(base_config.enemies_to_clear * adjustment),
            boss_type=base_config.boss_type,
            mines_enabled=base_config.mines_enabled,
            formations_enabled=base_config.formations_enabled,
            formation_types=base_config.formation_types,
            theme_name=base_config.theme_name,
        )
    
    def get_skill_level(self) -> str:
        """Estima nível de habilidade do jogador."""
        if self.highest_level_reached <= 3:
            return "Novato"
        elif self.highest_level_reached <= 10:
            avg_clear_rate = sum(s.clear_rate for s in self.level_stats.values()) / len(self.level_stats)
            return "Intermediário" if avg_clear_rate > 0.6 else "Aprendendo"
        elif self.highest_level_reached <= 20:
            return "Veterano"
        else:
            return "Mestre"
```
