## 1. **Pool de Explosões: Vazamento de Tipo**

### Problema Identificado
Na linha 62 de `explosion_pool.py`:

```python
def release(self, explosion: Explosion) -> None:
    if explosion in self.active:
        self.active.remove(explosion)
        explosion.time = 0
        explosion.particles.clear()
        explosion.explosion_type = None  # ← RESETAR TIPO!
        self.available.append(explosion)
```

**Por que é problema?** Se uma explosão do tipo `SLIME` é reutilizada, o tipo antigo pode vazar se não for resetado corretamente no `release()`. Seu código JÁ FAZ ISSO (comentário "ATUALIZAR ESTA LINHA"), então está correto! ✅

### Recomendação
Manter como está. Garantir que em **todos os `release()`** o tipo seja resetado:

```python
def release(self, explosion: Explosion) -> None:
    if explosion in self.active:
        self.active.remove(explosion)
        # Reset total: restaurar estado inicial do pool
        explosion.time = 0
        explosion.particles.clear()
        explosion.explosion_type = None  # ✅ Essencial
        self.available.append(explosion)
```

### Sugestão: Adicionar Efeitos de Morte Graduais

```python
class EnemyDeathEffect:
    """Efeito visual de desaparecimento do inimigo ao morrer."""
    
    def __init__(self, x: float, y: float, w: float, h: float, duration: float = 0.5):
        self.x, self.y = x, y
        self.w, self.h = w, h
        self.duration = duration
        self.elapsed = 0.0
        self.particles: list[list[float]] = []  # (x, y, vx, vy, life, color)
        self._create_debris()
    
    def _create_debris(self):
        """Cria pequenos fragmentos espalhados."""
        for _ in range(8):
            angle = random.uniform(0, 360)
            rad_angle = math.radians(angle)
            speed = random.uniform(100, 250)
            
            vx = speed * math.cos(rad_angle)
            vy = speed * math.sin(rad_angle)
            
            self.particles.append([
                self.x + random.uniform(-self.w/2, self.w/2),
                self.y + random.uniform(-self.h/2, self.h/2),
                vx, vy,
                self.duration,
                (200, 200, 200)  # Cinza
            ])
    
    def update(self, dt: float) -> bool:
        self.elapsed += dt
        for p in self.particles:
            p[0] += p[2] * dt
            p[1] += p[3] * dt
            p[2] *= 0.9
            p[3] *= 0.9
            p[4] -= dt
        self.particles = [p for p in self.particles if p[4] > 0]
        return self.elapsed >= self.duration
    
    def draw(self, screen: pygame.Surface):
        for p in self.particles:
            pygame.draw.circle(screen, (int(p[5][0]), int(p[5][1]), int(p[5][2])), 
                             (int(p[0]), int(p[1])), 2)
```

Depois adicionar ao `EntityManager`:

```python

        self.death_effects: list[EnemyDeathEffect] = []
    
    def spawn_death_effect(self, x: float, y: float, w: float, h: float):
        self.death_effects.append(EnemyDeathEffect(x, y, w, h))
    
    def update_death_effects(self, dt: float):
        finished = []
        for effect in self.death_effects:
            if effect.update(dt):
                finished.append(effect)
        for effect in finished:
            self.death_effects.remove(effect)
    
    def draw_death_effects(self, screen: pygame.Surface):
        for effect in self.death_effects:
            effect.draw(screen)
```

## 6. **Pool de Explosões: Dimensionamento Dinâmico**

### Problema Identificado
Pool inicializado com tamanho fixo:

```python
self.explosion_pool = ExplosionPool(initial_size=50)  # Hardcoded
```

Em fases intensas com muitos inimigos, pode haver:
- Pool vazio → criação de novas explosões (pior performance)
- Memory leak se muitas explosões deixarem de ser reutilizadas

### Sugestão: Pool Adaptativo

```python
class ExplosionPool:
    def __init__(self, initial_size: int = 50, max_size: int = 200):
        self.available: List[Explosion] = []
        self.active: List[Explosion] = []
        self.max_size = max_size
        self.created_count = initial_size  # Rastrear total criadas
        
        for _ in range(initial_size):
            explosion = Explosion(0, 0, size=30)
            explosion.time = 0
            explosion.particles.clear()
            self.available.append(explosion)
    
    def get(self, x: float, y: float, size: int = 30, 
            explosion_type: list[tuple[int, int, int]] | None = None) -> Explosion:
        """Obtém explosão do pool, criando se necessário (até limite)."""
        if self.available:
            explosion = self.available.pop()
            explosion.reset(x, y, size, explosion_type)
        else:
            # Pool vazio: criar nova se não atingiu limite
            if self.created_count < self.max_size:
                explosion = Explosion(x, y, size, explosion_type)
                self.created_count += 1
            else:
                # Limite atingido: reutilizar a mais antiga (fallback)
                if self.active:
                    explosion = self.active.pop(0)
                    explosion.reset(x, y, size, explosion_type)
                else:
                    explosion = Explosion(x, y, size, explosion_type)
                    self.created_count += 1
        
        self.active.append(explosion)
        return explosion
    
    def get_stats(self) -> Dict[str, Union[int, float]]:
        """Retorna estatísticas detalhadas."""
        total = len(self.active) + len(self.available)
        return {
            "active": len(self.active),
            "available": len(self.available),
            "total_created": self.created_count,
            "max_size": self.max_size,
            "utilization": len(self.active) / self.created_count if self.created_count > 0 else 0,
            "utilization_percent": f"{(len(self.active) / self.created_count * 100) if self.created_count > 0 else 0:.1f}%"
        }
```

**Benefícios:**
- ✅ Adapta-se automaticamente a fases intensas
- ✅ Evita memory leak
- ✅ Facilita debug via stats

```python
class DamageIndicator:
    """Indicador de dano flutuante."""
    def __init__(self, x: float, y: float, damage: int, color: tuple = (255, 0, 0)):
        self.x = x
        self.y = y
        self.damage = damage
        self.color = color
        self.life = 0.5  # segundos
        self.elapsed = 0.0
    
    def update(self, dt: float) -> bool:
        self.elapsed += dt
        self.y -= 50 * dt  # Flutua para cima
        return self.elapsed >= self.life
    
    def draw(self, screen: pygame.Surface, font: pygame.font.Font):
        alpha = 1.0 - (self.elapsed / self.life)
        if alpha > 0:
            text = font.render(f"-{self.damage}", True, self.color)
            # Aplicar alpha manualmente se necessário
            screen.blit(text, (int(self.x), int(self.y)))
```

## 8. **Sincronização de Som com Explosão**

### Problema Identificado


### Sugestão: Wrapper de Efeito de Morte

```python
class DeathEffect:
    """Agrupa explosão + som + score em um único efeito sincronizado."""
    def __init__(self, x: float, y: float, sound_id: str, 
                 explosion_size: int = 40, points: int = 100):
        self.x, self.y = x, y
        self.sound_id = sound_id
        self.explosion_size = explosion_size
        self.points = points
        self.triggered = False
    
    def trigger(self, entity_manager, sound_manager):
        """Dispara explosão + som sincronizados."""
        if not self.triggered:
            entity_manager.spawn_explosion(self.x, self.y, size=self.explosion_size)
            if self.sound_id == "asteroid":
                sound_manager.play_explosion_asteroid()
            elif self.sound_id == "alien":
                sound_manager.play_explosion_alien()
            self.triggered = True
```

```python
def diagnose_pool_health(pool: ExplosionPool, threshold: float = 0.8) -> str:
    """Diagnostica se o pool está dimensionado corretamente."""
    stats = pool.get_stats()
    utilization = stats["utilization"]
    
    if utilization > threshold:
        return f"⚠️ ALERTA: Pool sobrecarga! Utilização: {stats['utilization_percent']}"
```

```python
def _apply_boss_damage(self, projectiles, boss, floating_scores, entity_manager, ...):
```
