<script>
    document.addEventListener('DOMContentLoaded', () => {
        const enemy = document.getElementById('enemy');

        const neutralArmor = {
            outline: '#0b0c10',
            body:    '#3a4f63',
            dark:    '#263442',
            light:   '#637a8c',
        };

        const palettes = [
            {
                name: 'inferno',
                core: '#ffebb5', mid: '#ff8c00', outer: '#d12000',
                angryEye: '#ff8c00',
                robotOutline: '#3d1400', robotBody: '#5e473e',
                robotDark: '#42302a',    robotLight: '#8a6c62',
            },
            {
                name: 'toxina',
                core: '#eaffd0', mid: '#39ff14', outer: '#008000',
                angryEye: '#39ff14',
                robotOutline: '#0d2e0d', robotBody: '#47633a',
                robotDark: '#304226',    robotLight: '#7a8c63',
            },
            {
                name: 'nevasca',
                core: '#ffffff', mid: '#8ae2ff', outer: '#0055ff',
                angryEye: '#8ae2ff',
                robotOutline: '#001a33', robotBody: '#4d6e8c',
                robotDark: '#2b4a69',    robotLight: '#8ca8c4',
            }
        ];

        function applyPalette(p) {
            const vars = {
                '--energy-core':     p.core,
                '--energy-mid':      p.mid,
                '--energy-outer':    p.outer,
                '--eye-pupil-angry': p.angryEye,
                '--current-outline': p.robotOutline,
                '--current-body':    p.robotBody,
                '--current-dark':    p.robotDark,
                '--current-light':   p.robotLight,
                '--thruster-1':      p.core,
                '--thruster-2':      p.mid,
                '--thruster-3':      p.outer,
            };
            for (const [key, value] of Object.entries(vars)) {
                enemy.style.setProperty(key, value);
            }
        }

        function resetArmor() {
            enemy.style.setProperty('--current-outline', neutralArmor.outline);
            enemy.style.setProperty('--current-body',    neutralArmor.body);
            enemy.style.setProperty('--current-dark',    neutralArmor.dark);
            enemy.style.setProperty('--current-light',   neutralArmor.light);
            enemy.style.setProperty('--thruster-1', '#ffffff');
            enemy.style.setProperty('--thruster-2', '#5CE1E6');
            enemy.style.setProperty('--thruster-3', '#0077c2');
        }

        const wait = ms => new Promise(resolve => setTimeout(resolve, ms));

        async function attackCycle() {
            resetArmor();
            await wait(3000);

            const palette = palettes[Math.floor(Math.random() * palettes.length)];
            enemy.setAttribute('data-theme', palette.name);
            applyPalette(palette);
            enemy.classList.add('charging');
            await wait(1500);

            enemy.classList.remove('charging');

            const angle    = (Math.random() - 0.5) * (Math.PI / 1.5);
            const distance = 800;
            const recoil   = 18;
            const recoilDeg = -Math.sin(angle) * 8;

            enemy.style.setProperty('--shot-x',         `${Math.sin(angle)  * distance}px`);
            enemy.style.setProperty('--shot-y',         `${-Math.cos(angle) * distance}px`);
            enemy.style.setProperty('--recoil-x',       `${-Math.sin(angle) * recoil}px`);
            enemy.style.setProperty('--recoil-y',       `${Math.cos(angle)  * recoil}px`);
            enemy.style.setProperty('--recoil-rotate',  `${recoilDeg}deg`);

            enemy.classList.add('firing');
            await wait(500);

            enemy.classList.remove('firing');
            attackCycle();
        }

        attackCycle();
    });
</script>
