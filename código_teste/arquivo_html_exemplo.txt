<script>
const SCALE = 6;

const _ = null;
const A = '#2a2040'; const B = '#7b6b55'; const C = '#9c8566';
const D = '#b89d7a'; const E = '#d4b98a'; const F = '#e8d4a8';
const G = '#5b9fc8'; const H = '#9dd4f0'; const W = '#ffffff';
const R1 = '#ff4d4d'; const R2 = '#4a0e0e'; const R3 = '#ff9999';
const Y1 = '#ffb84d'; const Y2 = '#5c3a21';
const P1 = '#d942ff'; const P2 = '#4a0e80';
const L1 = '#00f2ff';

const map = [
  [_,_,_,_,_,_,_,_,_,F,_,_,_,_,_,_,_,_,_],
  [_,_,_,_,_,_,_,_,E,D,E,_,_,_,_,_,_,_,_],
  [_,_,_,_,_,_,_,D,C,B,C,D,_,_,_,_,_,_,_],
  [_,_,_,_,_,_,E,D,C,B,C,D,E,_,_,_,_,_,_],
  [_,_,_,_,_,F,E,D,C,B,C,D,E,F,_,_,_,_,_],
  [_,_,_,_,E,D,D,C,C,B,C,C,D,D,E,_,_,_,_],
  [_,_,_,D,C,C,C,B,B,A,B,B,C,C,C,D,_,_,_],
  [_,_,E,D,C,C,B,_,_,_,_,_,B,C,C,D,E,_,_],
  [_,_,E,D,C,C,B,B,B,B,B,B,B,C,C,D,E,_,_],
  [_,E,D,C,C,B,B,B,B,B,B,B,B,B,C,C,D,E,_],
  [E,D,C,C,B,B,B,B,B,A,B,B,B,B,B,C,C,D,E],
  [D,C,B,B,B,B,B,B,A,A,A,B,B,B,B,B,B,C,D],
  [C,B,B,B,B,B,B,A,A,A,A,A,B,B,B,B,B,B,C],
  [_,A,B,B,B,B,A,A,B,B,B,A,A,B,B,B,B,A,_],
  [_,_,A,B,B,A,A,B,B,B,B,B,A,A,B,B,A,_,_],
  [_,_,_,A,B,B,B,B,B,B,B,B,B,B,B,A,_,_,_],
  [_,_,_,_,A,B,B,B,B,B,B,B,B,B,A,_,_,_,_],
  [_,_,_,_,_,A,B,B,B,B,B,B,B,A,_,_,_,_,_],
  [_,_,_,_,_,_,A,B,B,B,B,B,A,_,_,_,_,_,_],
  [_,_,_,_,_,_,_,A,B,B,B,A,_,_,_,_,_,_,_],
  [_,_,_,_,_,_,_,_,A,B,A,_,_,_,_,_,_,_,_],
  [_,_,_,_,_,_,_,_,_,A,_,_,_,_,_,_,_,_,_],
];

const ROWS = map.length;
const COLS = map[0].length;
const canvas = document.getElementById('ship');
const ctx = canvas.getContext('2d');

canvas.width = (COLS + 100) * SCALE;
canvas.height = 800;

// FIX 3: constantes do thruster extraídas
const THRUSTER_RINGS = 5;
const THRUSTER_SPEED = 2.0;
const THRUSTER_MAX_DROP = 16 * SCALE;

// FIX 5: limite máximo de partículas de carga
const MAX_CHARGE_PARTICLES = 120;

let projectiles = [];
let chargeParticles = [];
let earthChunks = [];
let snakeOrbs = [];
let fxParticles = [];
let sandParticles = [];
let eyeGrowth = 0;
let currentFloatY = 0;
let jitterX = 0;
let jitterY = 0;
let sweepAngle = 0;

// FIX 4: variáveis de pupil compartilhadas entre física e render
let sharedScanStep = 0;
let sharedPupilX = 0;
let sharedCenterX = 0;
let sharedCenterY = 0;

const easeOutCubic = x => 1 - Math.pow(1 - x, 3);
const clamp = (val, min, max) => Math.min(Math.max(val, min), max);

// ----------------------------------------------------
// 1. MÁQUINA DE ESTADOS
// ----------------------------------------------------
const SM = {
  state: "SCAN",
  ticks: 0,
  // FIX 2: flag explícita para controlar disparo do laser
  chargeComplete: false,

  change(newState) {
    this.state = newState;
    this.ticks = 0;

    if (newState === "EARTH_SHAKE") initEarthAttack();
    if (newState === "ORB_SPAWN")   initOrbAttack();

    if (newState === "FIRE") {
      projectiles.push({ x: canvas.width / 2, y: 200 + 7.5 * SCALE + currentFloatY, angle: 0 });
      chargeParticles = [];
    }

    // FIX 2: reseta a flag ao entrar em CHARGE
    if (newState === "CHARGE") {
      this.chargeComplete = false;
    }

    // FIX 1: reseta sweepAngle ao iniciar a varredura
    if (newState === "SWEEP_FIRE") {
      const sweepTotal = 30 * Math.PI / 180;
      sweepAngle = Math.PI / 2 - sweepTotal / 2;
    }

    // FIX 6: transição de fase das pedras feita aqui, não no updatePhysics
    if (newState === "EARTH_ORBIT") {
      earthChunks.forEach(c => { c.phase = 'orbiting'; });
    }
    if (newState === "EARTH_FIRE") {
      earthChunks.forEach(c => {
        c.fireDelay = Math.random() * 40;
      });
    }
  },

  update(dt) {
    const timeSpeed = dt * 60;
    this.ticks += timeSpeed;

    switch (this.state) {
      case "SCAN":
        eyeGrowth = 0;
        if (this.ticks > 120) {
          const rand = Math.random();
          if      (rand < 0.25) this.change("OPENING");
          else if (rand < 0.50) this.change("EARTH_SHAKE");
          else if (rand < 0.75) this.change("ORB_SPAWN");
          else                  this.change("SWEEP_CHARGE");
        }
        break;

      // ATAQUE 1: LASER
      case "OPENING":
        eyeGrowth = easeOutCubic(clamp(this.ticks / 25, 0, 1));
        if (eyeGrowth >= 0.99 && this.ticks > 120) this.change("CHARGE");
        break;

      case "CHARGE":
        // FIX 2: usa flag em vez de checar array que pode estar vazio prematuramente
        if (this.ticks > 130 && !this.chargeComplete) {
          this.chargeComplete = true;
          this.change("FIRE");
        }
        break;

      case "FIRE":
        if (this.ticks > 25) this.change("CLOSING");
        break;

      // ATAQUE 2: TERRA
      case "EARTH_SHAKE":
        if (this.ticks > 50) this.change("EARTH_PULL");
        break;
      case "EARTH_PULL":
        if (this.ticks > 80) this.change("EARTH_ORBIT");
        break;
      case "EARTH_ORBIT":
        if (this.ticks > 90) this.change("EARTH_FIRE");
        break;
      case "EARTH_FIRE":
        if (earthChunks.length === 0) this.change("SCAN");
        break;

      // ATAQUE 3: ORBES
      case "ORB_SPAWN":
        eyeGrowth = easeOutCubic(clamp(this.ticks / 25, 0, 1));
        if (this.ticks > 60) this.change("ORB_HOLD");
        break;
      case "ORB_HOLD":
        if (this.ticks > 40) this.change("ORB_FIRE");
        break;
      case "ORB_FIRE":
        if (snakeOrbs.length === 0) this.change("CLOSING");
        break;

      // ATAQUE 4: LASER SWEEP
      case "SWEEP_CHARGE":
        eyeGrowth = easeOutCubic(clamp(this.ticks / 20, 0, 1));
        if (this.ticks > 80) this.change("SWEEP_FIRE");
        break;

      case "SWEEP_FIRE": {
        const sweepTotal = 30 * Math.PI / 180;
        const fireDuration = 45;
        const progress = clamp(this.ticks / fireDuration, 0, 1);
        // FIX 1: sweepAngle avança a partir do valor resetado em change()
        sweepAngle = (Math.PI / 2 - sweepTotal / 2) + (progress * sweepTotal);
        if (this.ticks > fireDuration + 10) this.change("CLOSING");
        break;
      }

      case "CLOSING":
        eyeGrowth = 1 - easeOutCubic(clamp(this.ticks / 30, 0, 1));
        if (eyeGrowth <= 0.01) this.change("SCAN");
        break;
    }
  }
};

function initEarthAttack() {
  earthChunks = [];
  for (let i = 0; i < 15; i++) {
    earthChunks.push({
      x: (canvas.width / 2) + (Math.random() - 0.5) * 400,
      y: canvas.height + 50 + Math.random() * 150,
      targetRadiusX: 100 + Math.random() * 60,
      targetRadiusY: 30 + Math.random() * 30,
      angle: Math.random() * Math.PI * 2,
      orbitSpeed: 0.03 + Math.random() * 0.04,
      // FIX 6: fase inicial sempre 'pulling'; orbiting é setado via SM.change()
      phase: 'pulling',
      size: Math.random() > 0.5 ? 2 : 3,
      color: [B, C, D][Math.floor(Math.random() * 3)],
      vx: 0, vy: 0, fireDelay: 0
    });
  }
}

function initOrbAttack() {
  snakeOrbs = [];
  const angles = [0, 45, 90, 135, 180, 225, 270, 315].map(a => a * Math.PI / 180);
  angles.forEach(angle => {
    snakeOrbs.push({
      angle, radius: 0, targetRadius: 18 * SCALE, phase: 'spawning',
      fireDelay: Math.random() * 80,
      waveFreq: 0.015 + Math.random() * 0.01,
      waveAmp: 25 + Math.random() * 15,
      wavePhase: Math.random() * Math.PI * 2,
      travelDist: 0, x: 0, y: 0, startX: 0, startY: 0
    });
  });
}

function updateShared(t) {
  // FIX 4: cálculo único de scanStep e pupilX, usado por física e render
  const isScanning = ["SCAN", "OPENING", "ORB_SPAWN"].includes(SM.state);
  sharedScanStep = isScanning ? Math.round(Math.cos(t * 3)) : 0;

  sharedCenterX = canvas.width / 2;
  sharedCenterY = 200 + 7.5 * SCALE + currentFloatY;

  const visorX = (sharedCenterX - (COLS / 2) * SCALE) + 7 * SCALE;
  sharedPupilX = visorX + (SCALE * 2.5) + (sharedScanStep * SCALE) + jitterX;
}

function updatePhysics(dt, t) {
  const timeSpeed = dt * 60;
  const centerX = sharedCenterX + jitterX;
  const centerY = sharedCenterY + jitterY;

  const isAnchored = [
    "CHARGE","FIRE","EARTH_SHAKE","EARTH_PULL","EARTH_ORBIT","EARTH_FIRE",
    "ORB_SPAWN","ORB_HOLD","ORB_FIRE","SWEEP_CHARGE","SWEEP_FIRE"
  ].includes(SM.state);

  let targetFloat = isAnchored ? 0 : Math.round(Math.sin(t * 2.5) * 12);
  currentFloatY += (targetFloat - currentFloatY) * (0.1 * timeSpeed);

  jitterX = 0; jitterY = 0;
  if (SM.state === "EARTH_SHAKE" || SM.state === "SWEEP_FIRE") {
    jitterX = (Math.random() - 0.5) * SCALE;
    jitterY = (Math.random() - 0.5) * SCALE;
  }

  // Areia
  if (Math.random() < 0.1 * timeSpeed) {
    sandParticles.push({
      x: centerX + (Math.random() - 0.5) * (COLS * SCALE * 0.6),
      y: centerY + (ROWS * SCALE * 0.4),
      vx: (Math.random() - 0.5) * 0.4,
      vy: 0.8 + Math.random() * 1.2,
      size: SCALE / 3,
      color: [C, D, E][Math.floor(Math.random() * 3)],
      life: 1
    });
  }
  for (let i = sandParticles.length - 1; i >= 0; i--) {
    let s = sandParticles[i];
    s.x += s.vx * timeSpeed; s.y += s.vy * timeSpeed; s.life -= 0.008 * timeSpeed;
    if (s.life <= 0) sandParticles.splice(i, 1);
  }

  // FX Genérico
  for (let i = fxParticles.length - 1; i >= 0; i--) {
    let p = fxParticles[i];
    p.x += p.vx * timeSpeed; p.y += p.vy * timeSpeed; p.life -= 0.05 * timeSpeed;
    if (p.life <= 0) fxParticles.splice(i, 1);
  }

  // Pedras
  for (let i = earthChunks.length - 1; i >= 0; i--) {
    let c = earthChunks[i];
    if (c.phase === 'pulling' || c.phase === 'orbiting') {
      c.angle += c.orbitSpeed * timeSpeed;
      const lerpSpeed = c.phase === 'orbiting' ? 0.2 : 0.05;
      const tx = centerX + Math.cos(c.angle) * c.targetRadiusX;
      const ty = centerY + Math.sin(c.angle) * c.targetRadiusY;
      c.x += (tx - c.x) * lerpSpeed * timeSpeed;
      c.y += (ty - c.y) * lerpSpeed * timeSpeed;
      // FIX 6: EARTH_FIRE agora só ativa fireDelay (já setado em SM.change)
      if (SM.state === "EARTH_FIRE") {
        if (c.fireDelay <= 0) {
          c.phase = 'firing';
          c.vx = (Math.random() - 0.5) * 16;
          c.vy = 12 + Math.random() * 15;
        } else {
          c.fireDelay -= timeSpeed;
        }
      }
    } else if (c.phase === 'firing') {
      c.x += c.vx * timeSpeed; c.y += c.vy * timeSpeed; c.angle += 0.2 * timeSpeed;
    }
    if (c.y > canvas.height + 200) earthChunks.splice(i, 1);
  }

  // Orbes Serpente
  for (let i = snakeOrbs.length - 1; i >= 0; i--) {
    let o = snakeOrbs[i];
    if (o.phase === 'spawning') {
      o.radius += (o.targetRadius - o.radius) * 0.1 * timeSpeed;
      o.x = centerX + Math.cos(o.angle) * o.radius;
      o.y = centerY + Math.sin(o.angle) * o.radius;
      if (SM.state === "ORB_HOLD") o.phase = 'holding';
    } else if (o.phase === 'holding') {
      o.x = centerX + Math.cos(o.angle) * o.radius;
      o.y = centerY + Math.sin(o.angle) * o.radius;
      if (SM.state === "ORB_FIRE") {
        o.fireDelay -= timeSpeed;
        if (o.fireDelay <= 0) {
          o.phase = 'firing'; o.startX = o.x; o.startY = o.y;
          for (let p = 0; p < 6; p++) {
            fxParticles.push({
              x: o.x, y: o.y,
              vx: -Math.cos(o.angle) * (Math.random() * 4) + (Math.random() - 0.5) * 2,
              vy: -Math.sin(o.angle) * (Math.random() * 4) + (Math.random() - 0.5) * 2,
              life: 1,
              color: Math.random() > 0.5 ? P1 : W,
              size: Math.random() > 0.5 ? SCALE / 2 : SCALE
            });
          }
        }
      }
    } else if (o.phase === 'firing') {
      o.travelDist += 5 * timeSpeed;
      const bx = o.startX + Math.cos(o.angle) * o.travelDist;
      const by = o.startY + Math.sin(o.angle) * o.travelDist;
      const offset = Math.sin(o.travelDist * o.waveFreq + o.wavePhase) * o.waveAmp;
      o.x = bx + Math.cos(o.angle + Math.PI / 2) * offset;
      o.y = by + Math.sin(o.angle + Math.PI / 2) * offset;
    }
    if (o.y > canvas.height + 100 || o.y < -100 || o.x < -100 || o.x > canvas.width + 100) {
      snakeOrbs.splice(i, 1);
    }
  }

  // FIX 5: partículas de carga com limite máximo
  if (SM.state === "CHARGE" && SM.ticks < 120 && chargeParticles.length < MAX_CHARGE_PARTICLES) {
    for (let i = 0; i < 4; i++) {
      chargeParticles.push({
        angle: Math.random() * Math.PI * 2,
        dist: 120 + Math.random() * 40,
        startDist: 120,
        orbitSpeed: 0.05 + Math.random() * 0.1,
        radialSpeed: 2 + Math.random() * 2,
        color: Math.random() > 0.5 ? R1 : R3,
        size: 2 + Math.random() * 3
      });
    }
  }

  // FIX 4: usa sharedPupilX calculado em updateShared
  for (let i = chargeParticles.length - 1; i >= 0; i--) {
    let p = chargeParticles[i];
    p.dist -= p.radialSpeed * timeSpeed;
    p.angle += p.orbitSpeed * (1 + (1 - p.dist / p.startDist) * 2) * timeSpeed;
    if (p.dist < 2) { chargeParticles.splice(i, 1); continue; }
    p.x = sharedPupilX + Math.cos(p.angle) * p.dist;
    p.y = (sharedCenterY + jitterY) + Math.sin(p.angle) * p.dist;
  }

  projectiles.forEach(p => { p.y += 18 * timeSpeed; p.angle += 0.5 * timeSpeed; });
  projectiles = projectiles.filter(p => p.y <= canvas.height + 100);
}

function drawPixel(x, y, color, sX = SCALE, sY = SCALE) {
  ctx.fillStyle = color;
  ctx.fillRect(Math.round(x), Math.round(y), Math.round(sX), Math.round(sY));
}

function drawPixelCircle(cx, cy, radiusIdx, colorMain, colorCore, angle = 0) {
  const s = SCALE;
  if (radiusIdx === 0) {
    drawPixel(cx - s / 2, cy - s / 2, colorMain, s, s);
  } else if (radiusIdx === 1) {
    drawPixel(cx - s / 2, cy - s * 1.5, colorMain, s, s * 3);
    drawPixel(cx - s * 1.5, cy - s / 2, colorMain, s * 3, s);
    drawPixel(cx - s / 2, cy - s / 2, colorCore, s, s);
  } else {
    drawPixel(cx - s * 0.5, cy - s * 2.5, colorMain, s, s * 5);
    drawPixel(cx - s * 2.5, cy - s * 0.5, colorMain, s * 5, s);
    drawPixel(cx - s * 1.5, cy - s * 1.5, colorMain, s * 3, s * 3);
    for (let i = 0; i < 4; i++) {
      const rotA = angle + (i * Math.PI / 2);
      drawPixel(cx - s / 2 + Math.round(Math.cos(rotA)) * s, cy - s / 2 + Math.round(Math.sin(rotA)) * s, colorCore, s, s);
    }
  }
}

function drawRock(x, y, size, color) {
  const s = SCALE;
  if (size === 2) {
    drawPixel(x, y, color, s * 2, s * 2);
  } else {
    drawPixel(x, y, color, s * 3, s * 2);
    drawPixel(x + s, y - s, color, s, s);
    drawPixel(x - s, y + s, color, s, s);
  }
}

function drawThruster(centerX, startY, t) {
  drawPixel(centerX - 1.5 * SCALE, startY, W, 3 * SCALE, 2 * SCALE);
  for (let i = 0; i < THRUSTER_RINGS; i++) {
    let phase = ((t * THRUSTER_SPEED) + (i / THRUSTER_RINGS)) % 1.0;
    let w = (10 * SCALE) * (1 - phase);
    let h = (4 * SCALE) * (1 - phase);
    let y = startY + (phase * THRUSTER_MAX_DROP) + SCALE;
    let color = phase < 0.15 ? W : (phase < 0.5 ? H : G);
    ctx.globalAlpha = Math.max(0, 1 - Math.pow(phase, 2));
    w = Math.max(SCALE, Math.round(w / SCALE) * SCALE);
    h = Math.max(SCALE, Math.round(h / SCALE) * SCALE);
    if (w <= 2 * SCALE) {
      drawPixel(centerX - w / 2, y - h / 2, color, w, h);
    } else {
      drawPixel(centerX - w / 2 + SCALE, y - h / 2, color, w - 2 * SCALE, SCALE);
      drawPixel(centerX - w / 2 + SCALE, y + h / 2 - SCALE, color, w - 2 * SCALE, SCALE);
      drawPixel(centerX - w / 2, y - h / 2 + SCALE, color, SCALE, h - 2 * SCALE);
      drawPixel(centerX + w / 2 - SCALE, y - h / 2 + SCALE, color, SCALE, h - 2 * SCALE);
    }
  }
  ctx.globalAlpha = 1;
}

function drawLaserSweep(cx, cy, angle) {
  const len = 1200;
  ctx.save();
  ctx.translate(cx, cy);
  ctx.rotate(angle);
  ctx.globalAlpha = 0.3;
  drawPixel(0, -SCALE * 1.5, L1, len, SCALE * 3);
  ctx.globalAlpha = 0.8;
  drawPixel(0, -SCALE * 0.5, W, len, SCALE);
  ctx.restore();
  ctx.globalAlpha = 1;
}

function drawScene(t) {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  const isRed    = ["CHARGE", "FIRE"].includes(SM.state);
  const isEarth  = ["EARTH_SHAKE", "EARTH_PULL", "EARTH_ORBIT", "EARTH_FIRE"].includes(SM.state);
  const isPurple = ["ORB_SPAWN", "ORB_HOLD", "ORB_FIRE"].includes(SM.state);
  const isSweep  = ["SWEEP_CHARGE", "SWEEP_FIRE"].includes(SM.state);

  // FIX 4: usa coordenadas calculadas em updateShared
  const centerX = sharedCenterX + jitterX;
  const centerY = sharedCenterY + jitterY;
  const startX  = (sharedCenterX - (COLS / 2) * SCALE) + jitterX;
  const pupilX  = sharedPupilX;
  const eyeY    = centerY - (SCALE * 1.5);

  earthChunks.forEach(c => {
    if (Math.sin(c.angle) < 0 && c.phase !== 'firing') drawRock(c.x, c.y, c.size, c.color);
  });

  drawThruster(centerX, 200 + (ROWS + 2) * SCALE + currentFloatY + jitterY, t);

  let bgColor = W; let irisColor = H;
  if (isRed)    { bgColor = R2; irisColor = R1; }
  else if (isEarth)  { bgColor = Y2; irisColor = Y1; }
  else if (isPurple) { bgColor = P2; irisColor = P1; }
  else if (isSweep)  { bgColor = '#003333'; irisColor = L1; }

  const visorX = startX + 7 * SCALE - jitterX; // visor relativo ao body, não ao jitter do olho
  drawPixel(visorX + jitterX, eyeY, bgColor, SCALE * 5, SCALE * 3);
  drawPixel(visorX + jitterX + SCALE + (sharedScanStep * SCALE), eyeY, irisColor, SCALE * 3, SCALE * 3);
  drawPixel(visorX + jitterX + (SCALE * 2) + (sharedScanStep * SCALE), centerY - (SCALE / 2), W, SCALE, SCALE);

  sandParticles.forEach(p => { ctx.globalAlpha = p.life; drawPixel(p.x, p.y, p.color, p.size, p.size); });
  ctx.globalAlpha = 1;

  for (let r = 0; r < ROWS; r++) {
    for (let c = 0; c < COLS; c++) {
      const color = map[r][c];
      if (!color) continue;
      const offset = Math.round(eyeGrowth * SCALE);
      if (c >= 7 && c <= 11) {
        if (r === 7) continue;
        if (r === 6) { drawPixel(startX + c * SCALE, 200 + r * SCALE + currentFloatY + jitterY - offset, color); continue; }
        if (r === 8) { drawPixel(startX + c * SCALE, 200 + r * SCALE + currentFloatY + jitterY + offset, color); continue; }
      }
      drawPixel(startX + c * SCALE, 200 + r * SCALE + currentFloatY + jitterY, color);
    }
  }

  earthChunks.forEach(c => {
    if (Math.sin(c.angle) >= 0 || c.phase === 'firing') drawRock(c.x, c.y, c.size, c.color);
  });

  fxParticles.forEach(p => { ctx.globalAlpha = Math.max(0, p.life); drawPixel(p.x - p.size / 2, p.y - p.size / 2, p.color, p.size, p.size); });
  ctx.globalAlpha = 1;

  if (isSweep) {
    const sweepTotal = 30 * Math.PI / 180;
    const sweepStart = Math.PI / 2 - sweepTotal / 2;
    const sweepEnd   = Math.PI / 2 + sweepTotal / 2;
    ctx.save();
    ctx.globalAlpha = SM.state === "SWEEP_CHARGE" ? 0.15 + Math.sin(t * 20) * 0.1 : 0.15;
    ctx.fillStyle = L1;
    ctx.beginPath();
    ctx.moveTo(pupilX, centerY);
    ctx.arc(pupilX, centerY, 1200, sweepStart, sweepEnd);
    ctx.closePath();
    ctx.fill();
    ctx.restore();
  }

  if (SM.state === "SWEEP_FIRE") drawLaserSweep(pupilX, centerY, sweepAngle);

  chargeParticles.forEach(p => { ctx.globalAlpha = Math.min(1, (p.dist / 120) * 2); drawPixel(p.x, p.y, p.color, p.size, p.size); });
  ctx.globalAlpha = 1;

  if (SM.state === "CHARGE")       drawPixelCircle(pupilX, centerY, (SM.ticks > 90 ? 2 : (SM.ticks > 40 ? 1 : 0)), R1, W, t * 10);
  if (SM.state === "SWEEP_CHARGE") drawPixelCircle(pupilX, centerY, (SM.ticks > 40 ? 1 : 0), L1, W, t * 20);

  snakeOrbs.forEach(o => {
    if (o.phase === 'firing') {
      ctx.globalAlpha = 0.4;
      const trailX = o.x - Math.cos(o.angle) * SCALE * 1.5;
      const trailY = o.y - Math.sin(o.angle) * SCALE * 1.5;
      drawPixelCircle(trailX, trailY, 0, P1, P1);
      ctx.globalAlpha = 1;
    }
    drawPixelCircle(o.x, o.y, 1, P1, W, t * 5);
  });

  projectiles.forEach(p => { drawPixelCircle(p.x, p.y, 2, R1, W, p.angle); });
}

// 4. LOOP PRINCIPAL
// ----------------------------------------------------
let lastTs = 0;
function loop(ts) {
  if (!lastTs) lastTs = ts;
  // FIX 3 (dt cap): evita salto físico ao voltar de tab em background
  const dt = Math.min((ts - lastTs) / 1000, 0.05);
  lastTs = ts;

  SM.update(dt);
  // FIX 4: atualiza coordenadas compartilhadas antes de física e render
  updateShared(ts / 1000);
  updatePhysics(dt, ts / 1000);
  drawScene(ts / 1000);
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
